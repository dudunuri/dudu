        let currentGamePin = "";
        let players = []; // { id, name, score, currentAnswer, answerTime }
        let currentQIndex = 0;
        let timerInterval = null;
        let timeLeft = 30;
        let isHost = false;
        let myPlayerId = "";
        let broadcastChannel = null;

        // Initialize Cross-Tab Sync via BroadcastChannel
        function initBroadcast() {
            broadcastChannel = new BroadcastChannel("nav_quiz_game");
            broadcastChannel.onmessage = (event) => {
                const data = event.data;
                if (!data) return;

                if (isHost) {
                    handleHostMessage(data);
                } else {
                    handlePlayerMessage(data);
                }
            };
        }

        initBroadcast();

        function startHostMode() {
            isHost = true;
            currentGamePin = Math.floor(100000 + Math.random() * 900000).toString();
            
            document.getElementById("modeSelectView").classList.add("hidden");
            document.getElementById("hostLobbyView").classList.remove("hidden");
            document.getElementById("sessionBadge").classList.remove("hidden");
            document.getElementById("gamePinDisplay").innerText = currentGamePin;
            document.getElementById("lobbyPinDisplay").innerText = currentGamePin;

            // Generate QR Code
            const currentUrl = window.location.href.split('?')[0] + "?pin=" + currentGamePin;
            new QRCode(document.getElementById("qrcode"), {
                text: currentUrl,
                width: 160,
                height: 160,
                colorDark: "#0f172a",
                colorLight: "#ffffff"
            });
        }

        function handleHostMessage(data) {
            if (data.type === "PLAYER_JOIN") {
                if (data.pin === currentGamePin) {
                    if (!players.find(p => p.id === data.player.id)) {
                        players.push({
                            id: data.player.id,
                            name: data.player.name,
                            score: 0,
                            currentAnswer: null,
                            answerTime: 0
                        });
                        updateLobbyUI();
                    }
                }
            } else if (data.type === "SUBMIT_ANSWER") {
                const player = players.find(p => p.id === data.playerId);
                if (player && player.currentAnswer === null) {
                    player.currentAnswer = data.answerIndex;
                    player.answerTime = data.timeTaken;
                    updateAnsweredCount();
                }
            }
        }

        function updateLobbyUI() {
            document.getElementById("playerCount").innerText = players.length;
            const list = document.getElementById("lobbyPlayersList");
            list.innerHTML = players.map(p => `
                <span class="px-3 py-1.5 bg-indigo-950 border border-indigo-500/30 text-indigo-200 rounded-lg text-sm font-bold flex items-center gap-2 animate-pulse">
                    <i class="fa-solid fa-user text-xs text-indigo-400"></i> ${p.name}
                </span>
            `).join('');
        }

        function startQuiz() {
            if (players.length === 0) {
                alert("אנא המתן להתחברות לפחות צוער אחד לפני התחלת המשחק!");
                return;
            }
            currentQIndex = 0;
            document.getElementById("hostLobbyView").classList.add("hidden");
            document.getElementById("hostQuestionView").classList.remove("hidden");
            showQuestion(currentQIndex);
        }

        function showQuestion(index) {
            const q = quizData[index];
            document.getElementById("questionNumBadge").innerText = `שאלה ${index + 1} מתוך ${quizData.length}`;
            document.getElementById("hostQuestionText").innerText = q.q;
            document.getElementById("hostExplanationBox").classList.add("hidden");

            // Reset player answers
            players.forEach(p => { p.currentAnswer = null; p.answerTime = 0; });
            updateAnsweredCount();

            // Render options
            const colors = ['bg-red-600', 'bg-blue-600', 'bg-yellow-600', 'bg-emerald-600'];
            const icons = ['fa-square', 'fa-circle', 'fa-triangle-exclamation', 'fa-diamond'];
            
            const grid = document.getElementById("hostOptionsGrid");
            grid.innerHTML = q.opts.map((opt, i) => `
                <div id="optBox-${i}" class="${colors[i]} p-5 rounded-2xl text-white font-bold text-lg flex items-center justify-between shadow-lg opacity-90 transition">
                    <span>${String.fromCharCode(1504 + i)}. ${opt}</span>
                    <i class="fa-solid ${icons[i]} text-2xl opacity-40"></i>
                </div>
            `).join('');

            // Broadcast to players
            broadcastChannel.postMessage({
                type: "START_QUESTION",
                qIndex: index,
                opts: q.opts
            });

            // Start Timer
            timeLeft = 30;
            document.getElementById("timerDisplay").innerText = timeLeft;
            clearInterval(timerInterval);
            timerInterval = setInterval(() => {
                timeLeft--;
                document.getElementById("timerDisplay").innerText = timeLeft;
                if (timeLeft <= 0) {
                    clearInterval(timerInterval);
                    endQuestionTime();
                }
            }, 1000);
        }

        function updateAnsweredCount() {
            const answered = players.filter(p => p.currentAnswer !== null).length;
            document.getElementById("answeredCount").innerText = `${answered} / ${players.length}`;
            if (answered === players.length && players.length > 0) {
                clearInterval(timerInterval);
                endQuestionTime();
            }
        }

        function endQuestionTime() {
            const q = quizData[currentQIndex];

            // Calculate Scores
            players.forEach(p => {
                if (p.currentAnswer === q.correct) {
                    // Score formula based on speed (max 1000 pts)
                    const speedBonus = Math.round((p.answerTime / 30) * 500);
                    p.score += (500 + speedBonus);
                }
            });

            // Highlight Correct Answer on Host Screen
            q.opts.forEach((_, i) => {
                const box = document.getElementById(`optBox-${i}`);
                if (i === q.correct) {
                    box.classList.add("ring-4", "ring-white", "scale-105");
                } else {
                    box.classList.add("opacity-30");
                }
            });

            // Show Explanation
            document.getElementById("hostExplanationText").innerText = q.exp;
            document.getElementById("hostExplanationBox").classList.remove("hidden");

            // Broadcast End to Players
            broadcastChannel.postMessage({
                type: "END_QUESTION",
                correctIndex: q.correct,
                scores: players.map(p => ({ id: p.id, score: p.score }))
            });
        }

        function nextStep() {
            // Check if leaderboard should be shown (every 5 questions or last question)
            if ((currentQIndex + 1) % 5 === 0 || currentQIndex === quizData.length - 1) {
                showLeaderboard();
            } else {
                currentQIndex++;
                showQuestion(currentQIndex);
            }
        }

        function showLeaderboard() {
            document.getElementById("hostQuestionView").classList.add("hidden");
            document.getElementById("leaderboardView").classList.remove("hidden");

            // Sort Players
            const sorted = [...players].sort((a, b) => b.score - a.score).slice(0, 5);
            
            const isFinal = currentQIndex === quizData.length - 1;
            document.getElementById("leaderboardTag").innerText = isFinal ? "תוצאות סופיות!" : "טבלת מובילים זמנית";

            const medals = ['🥇', '🥈', '🥉', '4️⃣', '5️⃣'];
            document.getElementById("leaderboardList").innerHTML = sorted.map((p, i) => `
                <div class="flex items-center justify-between bg-slate-900 border border-slate-700 p-4 rounded-2xl">
                    <div class="flex items-center gap-3">
                        <span class="text-2xl">${medals[i]}</span>
                        <span class="font-bold text-white text-lg">${p.name}</span>
                    </div>
                    <span class="font-mono font-black text-emerald-400 text-xl">${p.score} נק'</span>
                </div>
            `).join('');

            if (isFinal) {
                confetti({ particleCount: 150, spread: 80, origin: { y: 0.6 } });
            }
        }

        function continueFromLeaderboard() {
            document.getElementById("leaderboardView").classList.add("hidden");
            if (currentQIndex < quizData.length - 1) {
                currentQIndex++;
                document.getElementById("hostQuestionView").classList.remove("hidden");
                showQuestion(currentQIndex);
            } else {
                alert("המבחן הסתיים בהצלחה! כל הכבוד לכל הצוערים.");
                location.reload();
            }
        }

        function startPlayerMode() {
            isHost = false;
            document.getElementById("modeSelectView").classList.add("hidden");
            document.getElementById("playerJoinView").classList.remove("hidden");

            // Auto fill PIN if in URL
            const urlParams = new URLSearchParams(window.location.search);
            if (urlParams.has("pin")) {
                document.getElementById("inputPin").value = urlParams.get("pin");
            }
        }

        function handlePlayerJoin(e) {
            e.preventDefault();
            const pin = document.getElementById("inputPin").value.trim();
            const name = document.getElementById("inputName").value.trim();

            myPlayerId = "p_" + Math.random().toString(36).substr(2, 9);
            document.getElementById("playerNameTag").innerText = name;

            broadcastChannel.postMessage({
                type: "PLAYER_JOIN",
                pin: pin,
                player: { id: myPlayerId, name: name }
            });

            document.getElementById("playerJoinView").classList.add("hidden");
            document.getElementById("playerWaitingView").classList.remove("hidden");
        }

        function handlePlayerMessage(data) {
            if (data.type === "START_QUESTION") {
                document.getElementById("playerWaitingView").classList.add("hidden");
                document.getElementById("playerFeedbackView").classList.add("hidden");
                document.getElementById("playerAnswerView").classList.remove("hidden");

                document.getElementById("playerQIndex").innerText = `שאלה ${data.qIndex + 1}`;
                
                // Set Option Button Text
                data.opts.forEach((opt, i) => {
                    document.querySelectorAll(`.opt-text-${i}`).forEach(el => el.innerText = opt);
                });

                window.questionStartTime = Date.now();
            } else if (data.type === "END_QUESTION") {
                document.getElementById("playerAnswerView").classList.add("hidden");
                document.getElementById("playerFeedbackView").classList.remove("hidden");

                // Update My Score
                const myScoreObj = data.scores.find(s => s.id === myPlayerId);
                if (myScoreObj) {
                    document.getElementById("playerScoreTag").innerText = `${myScoreObj.score} נקודות`;
                }

                if (window.mySelectedAnswer === data.correctIndex) {
                    document.getElementById("feedbackIcon").innerHTML = "🎉";
                    document.getElementById("feedbackTitle").innerText = "תשובה נכונה!";
                    document.getElementById("feedbackTitle").className = "text-3xl font-black text-emerald-400";
                    document.getElementById("feedbackPoints").innerText = "+ נקודות מהירות נוספו";
                } else if (window.mySelectedAnswer !== null) {
                    document.getElementById("feedbackIcon").innerHTML = "❌";
                    document.getElementById("feedbackTitle").innerText = "תשובה שגויה";
                    document.getElementById("feedbackTitle").className = "text-3xl font-black text-red-400";
                    document.getElementById("feedbackPoints").innerText = "לא נורא, בהצלחה בשאלה הבאה!";
                } else {
                    document.getElementById("feedbackIcon").innerHTML = "⏰";
                    document.getElementById("feedbackTitle").innerText = "נגמר הזמן!";
                    document.getElementById("feedbackTitle").className = "text-3xl font-black text-yellow-400";
                    document.getElementById("feedbackPoints").innerText = "לא נשלחה תשובה בזמן.";
                }

                window.mySelectedAnswer = null;
            }
        }

        function submitAnswer(ansIndex) {
            window.mySelectedAnswer = ansIndex;
            const timeTaken = Math.max(1, Math.floor((30000 - (Date.now() - window.questionStartTime)) / 1000));

            broadcastChannel.postMessage({
                type: "SUBMIT_ANSWER",
                playerId: myPlayerId,
                answerIndex: ansIndex,
                timeTaken: timeTaken
            });

            document.getElementById("playerAnswerView").classList.add("hidden");
            document.getElementById("playerFeedbackView").classList.remove("hidden");
            document.getElementById("feedbackIcon").innerHTML = "👍";
            document.getElementById("feedbackTitle").innerText = "התשובה נשלחה!";
            document.getElementById("feedbackTitle").className = "text-3xl font-black text-indigo-400";
            document.getElementById("feedbackPoints").innerText = "ממתין לסיום הטיימר...";
        }
